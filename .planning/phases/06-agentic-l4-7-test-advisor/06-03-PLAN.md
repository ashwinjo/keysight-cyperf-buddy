    **docker-compose.yml** — Add the `agent` service block. Read the existing file first to understand current structure. Insert the new service AFTER the `api` service block:

    ```yaml
      # L4-7 Test Advisor agent (Gemini-powered recommendation service)
      agent:
        build:
          context: ./agent-service
          dockerfile: Dockerfile
        container_name: cyperf_agent_l47
        ports:
          - "8001:8001"
        environment:
          GEMINI_API_KEY: ${GEMINI_API_KEY:-}
          BACKEND_API_URL: http://api:8000
          BACKEND_TIMEOUT_SECONDS: "10"
          LOG_LEVEL: ${LOG_LEVEL:-INFO}
          ENVIRONMENT: production
        depends_on:
          - api
        networks:
          - cyperf_network
        restart: unless-stopped
    ```

    CRITICAL: `GEMINI_API_KEY: ${GEMINI_API_KEY:-}` — the `:-` default means Docker Compose will not fail if the var is unset; instead the agent container itself handles the fail-fast at startup (as designed in config.py). This prevents `docker compose up` from aborting before the agent even starts.

    Do NOT change any existing service definitions. Do NOT change ports for existing services (postgres, redis, api=8000, frontend=5174).

    ```python
    """Unit tests for ranking logic and Pydantic models."""
    import pytest
    from ranking import rank_profiles_hybrid, RankedProfile
    from models import L47ScenarioRequest, L47RecommendationResponse, Recommendation

    def test_rank_profiles_returns_top_k(sample_apps):
        """rank_profiles_hybrid returns at most top_k results."""
        ranked = rank_profiles_hybrid(sample_apps, "HTTP load balancer performance", top_k=2)
        assert len(ranked) <= 2
        assert all(isinstance(r, RankedProfile) for r in ranked)

    def test_rank_profiles_sorted_descending(sample_apps):
        """Results are sorted by score descending."""
        ranked = rank_profiles_hybrid(sample_apps, "HTTP traffic performance", top_k=3)
        scores = [r.score for r in ranked]
        assert scores == sorted(scores, reverse=True)

    def test_rank_profiles_http_scores_highest(sample_apps):
        """'HTTP Traffic' profile scores highest for HTTP scenario."""
        ranked = rank_profiles_hybrid(sample_apps, "HTTP load balancer performance test", top_k=3)
        if ranked:
            assert ranked[0].profile_name == "HTTP Traffic"

    def test_rank_profiles_empty_input():
        """Empty candidates returns empty list."""
        ranked = rank_profiles_hybrid([], "any scenario", top_k=3)
        assert ranked == []

    def test_l47_request_validation_missing_field():
        """L47ScenarioRequest raises ValidationError when required field missing."""
        from pydantic import ValidationError
        with pytest.raises(ValidationError):
            L47ScenarioRequest(
                testing_focus="app_performance",
                use_case="short",  # Too short (min_length=10)
                objectives="valid objectives text here",
                timeline="2 weeks",
            )

    def test_l47_request_valid_fields(sample_request_app):
        """L47ScenarioRequest accepts valid input."""
        req = L47ScenarioRequest(**sample_request_app)
        assert req.testing_focus == "app_performance"

    def test_recommendation_response_no_next_steps():
        """L47RecommendationResponse does not have next_steps field (per CONTEXT.md)."""
        resp = L47RecommendationResponse(success=True, message="ok", recommendations=[])
        assert not hasattr(resp, "next_steps"), "next_steps must NOT be in response (deferred per CONTEXT.md)"
    ```
