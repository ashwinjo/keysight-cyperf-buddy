### Task 2: Create `frontend/src/hooks/useContactForm.ts`

This hook encapsulates the API call, keeping the component clean.

```typescript
import { useState } from 'react';
import axios from 'axios';
import { ContactFormRequest, ContactFormResponse } from '../types/api';

const API_BASE = '/api';

interface UseContactFormReturn {
  submitForm: (data: ContactFormRequest) => Promise<ContactFormResponse>;
  isSubmitting: boolean;
  error: string | null;
  reset: () => void;
}

export function useContactForm(): UseContactFormReturn {
  const [isSubmitting, setIsSubmitting] = useState(false);
  const [error, setError] = useState<string | null>(null);

  const submitForm = async (data: ContactFormRequest): Promise<ContactFormResponse> => {
    setIsSubmitting(true);
    setError(null);
    try {
      const res = await axios.post<ContactFormResponse>(
        `${API_BASE}/contact/submit`,
        data,
        { timeout: 15000 }
      );
      return res.data;
    } catch (err: any) {
      const message =
        err?.response?.data?.detail || 'Submission failed. Please try again.';
      setError(message);
      throw new Error(message);
    } finally {
      setIsSubmitting(false);
    }
  };

  const reset = () => {
    setError(null);
  };

  return { submitForm, isSubmitting, error, reset };
}
```

**Success criteria:**
- Hook compiles with no TypeScript errors.
- `isSubmitting` is true during the `axios.post` call and false after resolution or rejection.
- `error` is set from the API detail field when available; generic message otherwise.

### Task 4: Create `frontend/src/components/contact/ContactFormSidebar.tsx`

The main component. Manages its own state machine: `idle | confirming | form | submitting | success | error`.

**Props interface:**

```typescript
interface ContactFormSidebarProps {
  cveId: string;
  testable: boolean;
  context: ContactContext;
  cvssScore: number | null;
  attackProfiles: string[];
  isOpen: boolean;
  onClose: () => void;
}
```

**Zod schema (client-side validation):**

```typescript
const contactSchema = z.object({
  first_name: z.string().min(1, 'First name is required').max(100),
  last_name: z.string().min(1, 'Last name is required').max(100),
  company: z.string().min(1, 'Company is required').max(200),
  email: z.string().email('Enter a valid email address'),
});
type ContactFormData = z.infer<typeof contactSchema>;
```

**Component behavior (state machine):**

1. **`idle`** — Sheet is closed. No UI rendered.

2. **`confirming`** — Sheet opens; `ConfirmDialog` shown first. User must click "Continue" to see the form. Pressing Escape or "Cancel" calls `onClose()`.

3. **`form`** — After confirmation, display the 4-field React Hook Form. Submit button is disabled while `isSubmitting` is true. On submit:
   - Merge form data with props (`cveId`, `context`, `testable`, `cvssScore`, `attackProfiles`).
   - Call `submitForm(payload)` from `useContactForm`.
   - On success → `success` state with email preview.
   - On error → show inline error message, remain in `form` state (let user retry).

4. **`success`** — Display:
   - Green checkmark icon (use `lucide-react` `CheckCircle2` which is already in the package).
   - Text: `"Email sent to the Keysight team."`
   - Email preview block: monospace `<pre>` or `whitespace-pre-wrap` div, dark background (`bg-luxury-bg`), showing full email body.
   - "Close" button that calls `onClose()`.
   - No auto-close timer (removed per clarity preference — user closes manually).

**Sheet implementation** — use `@radix-ui/react-dialog` directly (already installed). Render as a right-side drawer. Width: `w-[420px]` on desktop, `w-full` on mobile via Tailwind responsive classes. Apply the existing dark theme classes:
- Backdrop: `bg-black/50`
- Panel: `bg-luxury-bg border-l border-luxury-border`
- Title: `text-luxury-accent font-display`
